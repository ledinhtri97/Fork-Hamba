[STEP1] - Initilation 
# Clone the repository https://github.com/ledinhtri97/Hamba.git

$ git clone --recursive https://github.com/ledinhtri97/Hamba.git

# Register username and password from https://mano.is.tue.mpg.de/login.php to access models 
$ export USERNAME=...
$ export PASSWORD=...

$ cd Hamba
$ sh scripts/ai_env_installation.sh
# after run  sh scripts/ai_env_installation.sh . please logout and login again


[STEP2] - Download models and packages

$ cd Hamba
$ sh scripts/download_all.sh
$ docker-compose up -d

$ docker exec -it hamba bash
$ cd /workspace
$ sh scripts/docker_install.sh


[STEP3] - Test with new input data

$ cd Hamba
$ mkdir -p downloads/data_input/
# please push video/images need to test inside downloads/data_input/

$ docker exec -it hamba bash
$ cd /workspace/

# test with video file
$ python demo.py --video_file downloads/data_input/video2.mp4 --out_folder downloads/data_output1 --full_frame

# test video images folder
$ python demo.py --img_folder downloads/data_input/sample_images1 --out_folder downloads/data_output2/ --full_frame
